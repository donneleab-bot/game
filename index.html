Нефритовая ] # Общие цвета для мелких деталей COLOR_BEAK = (255, 165, 0, 255) # Оранжевый для клюва и лапок COLOR_EYE = (0, 0, 0, 255) # Черный для глаз COLOR_SHADOW = (0, 0, 0, 50) # Полупрозрачный для теней def generate_chicken(body_color, crest_color): img = Image.new("RGBA", (16, 16), (0, 0, 0, 0)) pixels = img.load() for y in range(16): for x in range(16): char = chicken_template[y][x] if char == 'W': pixels[x, y] = body_color elif char == 'R': pixels[x, y] = crest_color elif char == 'O': pixels[x, y] = COLOR_BEAK elif char == 'B': pixels[x, y] = COLOR_EYE elif char == 'S': pixels[x, y] = COLOR_SHADOW return img def generate_chick(body_color): img = Image.new("RGBA", (8, 8), (0, 0, 0, 0)) pixels = img.load() for y in range(8): for x in range(8): char = chick_template[y][x] if char == 'Y': pixels[x, y] = body_color elif char == 'O': pixels[x, y] = COLOR_BEAK elif char == 'B': pixels[x, y] = COLOR_EYE return img # Цикл генерации всех 34 видов for i, (body, crest) in enumerate(palettes, 1): # Генерируем взрослую курицу (увеличим в 4 раза для четкого пиксель-арта при просмотре) chicken_img = generate_chicken(body, crest) chicken_resized = chicken_img.resize((64, 64), Image.Resampling.NEAREST) chicken_resized.save(f"chickens_and_chicks/chicken_{i:02d}.png") # Генерируем цыпленка chick_img = generate_chick(body) chick_resized = chick_img.resize((32, 32), Image.Resampling.NEAREST) chick_resized.save(f"chickens_and_chicks/chick_{i:02d}.png") print("Успешно сгенерировано 34 вида куриц и 34 вида цыплят в папке 'chickens_and_chicks'!")
